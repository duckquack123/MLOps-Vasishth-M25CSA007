#### Local Plotting Script

To generate train-val curves locally from WandB logs:

```python
# Generate from WandB API
import wandb
import matplotlib.pyplot as plt
import numpy as np

api = wandb.Api()
project = "YOUR_USERNAME/vit_cifar100_q1"

# Fetch baseline run
baseline_run = api.run(f"{project}/baseline_run_id")
baseline_history = baseline_run.history()

# Plot baseline
plt.figure(figsize=(15, 5))

plt.subplot(1, 3, 1)
plt.plot(baseline_history['epoch'], baseline_history['train_loss'], label='Train Loss')
plt.plot(baseline_history['epoch'], baseline_history['val_loss'], label='Val Loss')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.title('Baseline (no LoRA) - Loss Curves')
plt.legend()
plt.grid(True)

plt.subplot(1, 3, 2)
plt.plot(baseline_history['epoch'], baseline_history['train_acc'], label='Train Acc')
plt.plot(baseline_history['epoch'], baseline_history['val_acc'], label='Val Acc')
plt.xlabel('Epoch')
plt.ylabel('Accuracy')
plt.title('Baseline (no LoRA) - Accuracy Curves')
plt.legend()
plt.grid(True)

plt.subplot(1, 3, 3)
plt.plot(baseline_history['epoch'], baseline_history['lr'], label='Learning Rate')
plt.xlabel('Epoch')
plt.ylabel('Learning Rate')
plt.title('Baseline - Learning Rate Schedule')
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.savefig('q1_baseline_curves.png', dpi=300, bbox_inches='tight')
```

Shows accuracy degradation as epsilon (perturbation magnitude) increases:

### Option 2: Programmatic Access via WandB API

```python
import wandb
import matplotlib.pyplot as plt
from matplotlib.gridspec import GridSpec

# Initialize WandB API
api = wandb.Api()

# Fetch Q1 baseline run
q1_project = api.project("YOUR_USERNAME/vit_cifar100_q1")
baseline_run = q1_project.runs()[0]  # First run = baseline

# Get history
history = baseline_run.history()

# Create comprehensive figure
fig = plt.figure(figsize=(18, 10))
gs = GridSpec(3, 3, figure=fig)

# Plot 1: Baseline Loss
ax1 = fig.add_subplot(gs[0, 0])
ax1.plot(history['epoch'], history['train_loss'], label='Train Loss', linewidth=2)
ax1.plot(history['epoch'], history['val_loss'], label='Val Loss', linewidth=2)
ax1.set_xlabel('Epoch', fontsize=12)
ax1.set_ylabel('Loss', fontsize=12)
ax1.set_title('Q1 Baseline - Loss', fontsize=14, fontweight='bold')
ax1.legend()
ax1.grid(True, alpha=0.3)

# Plot 2: Baseline Accuracy
ax2 = fig.add_subplot(gs[0, 1])
ax2.plot(history['epoch'], history['train_acc'] * 100, label='Train Acc', linewidth=2)
ax2.plot(history['epoch'], history['val_acc'] * 100, label='Val Acc', linewidth=2)
ax2.set_xlabel('Epoch', fontsize=12)
ax2.set_ylabel('Accuracy (%)', fontsize=12)
ax2.set_title('Q1 Baseline - Accuracy', fontsize=14, fontweight='bold')
ax2.legend()
ax2.grid(True, alpha=0.3)

# Plot 3: Baseline Learning Rate
ax3 = fig.add_subplot(gs[0, 2])
ax3.plot(history['epoch'], history['lr'], linewidth=2, color='orange')
ax3.set_xlabel('Epoch', fontsize=12)
ax3.set_ylabel('Learning Rate', fontsize=12)
ax3.set_title('Q1 Baseline - Learning Rate', fontsize=14, fontweight='bold')
ax3.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('q1_comprehensive_visualization.png', dpi=300, bbox_inches='tight')
print("✅ Saved: q1_comprehensive_visualization.png")
```
